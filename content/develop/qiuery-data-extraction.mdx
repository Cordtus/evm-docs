# Extracting Cosmos Transaction Data

This tutorial demonstrates how to fetch and extract specific information from Cosmos blockchain transactions using REST API endpoints. We'll walk through a real example that queries transaction data for a specific address.

<ChainInfo chain="cosmos-evm" endpoint="https://rest.lavenderfive.com/cosmoshub" />

## The Transaction Extraction Script

Here's the complete script we'll use to fetch transactions:

<CodeCollapse language="javascript">
```javascript
import fetch from 'node-fetch';
import { setTimeout } from 'timers/promises';

async function fetchDataForAddress(address, messageType) {
    const baseUrl = 'https://rest.lavenderfive.com:443/cosmoshub';
    let page = 1;
    let hasNextPage = true;
    let allTransactions = [];
    
    while (hasNextPage) {
        try {
            // Use query parameter with proper syntax
            const url = `${baseUrl}/cosmos/tx/v1beta1/txs?query=message.sender='${address}' AND message.action='${messageType}'&pagination.limit=10&page=${page}`;
            
            console.log(`Fetching ${messageType} - Page ${page}`);
            
            const response = await fetch(url);
            
            if (!response.ok) {
                throw new Error(`Request failed with status ${response.status}`);
            }
            
            const data = await response.json();
            
            if (data.txs && data.txs.length > 0) {
                allTransactions.push(...data.txs);
                console.log(`Found ${data.txs.length} transactions on page ${page}`);
            }
            
            hasNextPage = Boolean(data.pagination && data.pagination.next_key);
            page++;
            
            // Limit to first 3 pages for testing
            if (page > 3) break;
            
            await setTimeout(2000);
            
        } catch (error) {
            console.error(`Error: ${error.message}`);
            await setTimeout(5000);
        }
    }
    
    return allTransactions;
}

async function main() {
    const address = 'cosmos1wev8ptzj27aueu04wgvvl4gvurax6rj5f0v7rw';
    const messageTypes = [
        '/cosmos.bank.v1beta1.MsgSend',
        '/ibc.core.client.v1.MsgUpdateClient',
        '/cosmos.bank.v1beta1.MsgMultiSend'
    ];
    
    let allResults = {};
    
    for (const messageType of messageTypes) {
        console.log(`Processing ${messageType}`);
        const transactions = await fetchDataForAddress(address, messageType);
        allResults[messageType] = transactions;
        await setTimeout(3000);
    }
    
    console.log(JSON.stringify(allResults, null, 2));
}

main().catch(error => console.error('Main Error:', error));
```
</CodeCollapse>

## Key API Discovery

The critical insight was discovering that the Cosmos Hub API requires:
- The `query` parameter instead of `events`
- Proper syntax: `message.sender='address' AND message.action='type'`
- The correct base URL: `https://rest.lavenderfive.com:443/cosmoshub`

## Extracting Specific Data

Now let's create helper functions to extract specific information from the transaction data:

<CodeCollapse language="javascript">
```javascript
// Extract transfer amount in ATOM (convert from uatom)
function getTransferAmount(transaction) {
    const message = transaction.body.messages[0];
    if (message['@type'] === '/cosmos.bank.v1beta1.MsgSend') {
        const atomAmount = message.amount.find(a => a.denom === 'uatom');
        if (atomAmount) {
            // Convert uatom to ATOM (1 ATOM = 1,000,000 uatom)
            return (parseInt(atomAmount.amount) / 1000000).toFixed(6) + ' ATOM';
        }
    }
    return null;
}

// Extract transaction fee in ATOM
function getTransactionFee(transaction) {
    const fee = transaction.auth_info.fee.amount.find(a => a.denom === 'uatom');
    if (fee) {
        return (parseInt(fee.amount) / 1000000).toFixed(6) + ' ATOM';
    }
    return null;
}

// Extract sender and recipient addresses
function getAddresses(transaction) {
    const message = transaction.body.messages[0];
    if (message['@type'] === '/cosmos.bank.v1beta1.MsgSend') {
        return {
            from: message.from_address,
            to: message.to_address
        };
    }
    return null;
}

// Extract gas information
function getGasInfo(transaction) {
    return {
        gasLimit: transaction.auth_info.fee.gas_limit,
        gasUsed: transaction.tx_response?.gasUsed || 'N/A'
    };
}

// Parse our actual transaction
const transaction = allResults['/cosmos.bank.v1beta1.MsgSend'][0];

console.log('Transfer Details:');
console.log('Amount:', getTransferAmount(transaction));
console.log('Fee:', getTransactionFee(transaction));
console.log('Addresses:', getAddresses(transaction));
console.log('Gas Info:', getGasInfo(transaction));
```
